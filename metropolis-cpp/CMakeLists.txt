cmake_minimum_required(VERSION 3.20)
project(MetropolisKayKit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
FetchContent_MakeAvailable(raylib)
add_executable(metropolis src/main.cpp)
target_link_libraries(metropolis PRIVATE raylib)
target_compile_options(metropolis PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic>)
set_target_properties(metropolis PROPERTIES OUTPUT_NAME "metropolis-kaykit")
